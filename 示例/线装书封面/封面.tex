% 现代蓝色线装书封面示例（issue #170）
% 中华书局、广陵书社等常见样式，参照 CY/T 247—2021《线装书籍要求》
% 编译：lualatex 封面.tex
\documentclass[四库全书]{ltc-guji}
\setmainfont{TW-Kai}
\无标点模式

\begin{document}

% 一、最简：只给书名，其余取默认
\线装书封面{史記}

% 二、书名 + 副题（签内小字）+ 作者 + 出版者
\线装书封面[
  副题 = 卷一至卷三,
  作者 = 漢　司馬遷　撰,
  出版者 = 中華書局,
]{史記}

% 三、长书名：书名签自动增高；作者位置手动调整
\线装书封面[
  副题 = 經部一,
  作者 = 清　紀昀等　撰,
  作者左距 = 50mm,
  作者上距 = 30mm,
  出版者 = 廣陵書社,
]{欽定四庫全書簡明目錄}

\end{document}
