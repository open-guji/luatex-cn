% 线装书封面示例（issue #170）
% 现代蓝色线装书封面（中华书局、广陵书社等影印古籍常见样式），参照 CY/T 247—2021《线装书籍要求》
% 编译：lualatex 封面.tex
\documentclass[四库全书]{ltc-guji}
\setmainfont{TW-Kai}
\无标点模式

\begin{document}

% 一、短书名：签条取最小高度，册次排在签条下部
\线装书封面[册次=第一冊]{史記}

% 二、四字书名 + 卷次
\线装书封面[卷次=卷一]{李白詩集}

% 三、长书名：签条随字数增高；册次、卷次两列并排
\线装书封面[册次=第九冊, 卷次=卷二十一至二十三]{欽定四庫全書簡明目錄}

% 四、小字紧随书名之后
\线装书封面[册次=第一冊, 小字位置=随后]{四書章句集注}

\end{document}
